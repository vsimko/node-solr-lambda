$ npm install solr-ts-client
typed-query.ts
import { createSolrClient }
  from "solr-ts-client";

interface Product {
  id: string;
  type: string;
  price: number;
}

const solr = createSolrClient({
  baseUrl: "http://localhost:8983",
  core: "gettingstarted"
});

const result = await solr
  .query<Product>({
    query: "type:book",
    limit: 20,
    sort: "price asc"
  });

// Fully typed response
result.response.docs;
